#include <windows.h>
#include <iostream>
#include <conio.h>
#include "winio.h"

#pragma comment(lib,"winio.lib")
using namespace std;

int ledout(unsigned short base, int port, unsigned char x)
{
	if (!SetPortVal(base + port, x, 1))
	{
		return 0;
	}
	return 1;
}

int main(void)
{
	unsigned short BASE_ADDRESS = 0x3F40;
	int iPort = 16;
	DWORD data = 0;
	unsigned char di_data;
	unsigned char led_data = 0;

	int mode = 1;
	int old1;
	int old2;
	int old3;

	int now1;
	int now2;
	int now3;

	int flag = 0;
	int n = 0;

	DWORD old_time;
	DWORD now_time;

	if (!InitializeWinIo())
	{
		cout << "Error In InitializeWinIo" << endl;
		return 1;
	}

	if (!GetPortVal(BASETYPES + iPort, &data, 1))
	{
		cout << "GetPortVal error!" << endl;
		ShutdownWinIo();
		return 1;
	}

	di_data = (unsigned char)(data & 0xFF);

	old1 = di_data & 0x01;
	old2 = (di_data >> 1) & 0x01;
	old3 = (di_data >> 2) & 0x01;

	old_time = GetTickCount();

	while (!_kbhit())
	{
		data = 0;
		GetPortVal(BASE_ADDRESS + iPort, &data, 1);
		di_data = (unsigned char)(data & 0xFF);
		now1 = di_data & 0x01;
		now2 = (di_data >> 1) & 0x01;
		now3 = (di_data >> 2) & 0x01;

		if (old1 == 1 && now1 == 0)
		{
			mode = 1;
			flag = 0;
			n = 0;
			old_time = GetTickCount();
			ledout(BASE_ADDRESS, iPort, 0x00);
		}
		else if (old2 == 1 && now2 == 0)
		{
			mode = 2;
			flag = 0;
			n = 0;
			old_time = GetTickCount();
			ledout(BASE_ADDRESS, iPort, 0x00);
		}
		else if (old3 == 1 && now3 == 0)
		{
			mode = 3;
			flag = 0;
			n = 0;
			old_time = GetTickCount();
			ledout(BASE_ADDRESS, iPort, 0x00);
		}
		old1 = now1;
		old2 = now2;
		old3 = now3;

		now_time = GetTickCount();
		if (mode == 1)
		{
			if (now_time - old_time >= 500)
			{
				old_time = now_time;
				flag = !flag;
				if (flag == 1)
				{
					led_data = 0xFF;
				}
				else
				{
					led_data = 0x00;
				}
				ledout(BASE_ADDRESS, iPort, led_data);
			}
		}

		else if (mode == 2)
		{
			if (now_time - old_time >= 150)
			{
				old_time = now_time;
				led_data = (unsigned char)(1 << n);
				ledout(BASE_ADDRESS, iPort, led_data);
				n++;
				if (n >= 8)
				{
					n = 0;
				}
			}
		}
		else if (mode == 3)
		{
			if (now_time - old_time >= 300)
			{
				old_time = now_time;
				flag = !flag;
				if (flag == 1)
				{
					led_data = 0x55;
				}
				else
				{
					led_data = 0xAA;
				}
				ledout(BASE_ADDRESS, iPort, led_data);
			}
		}
		Sleep(10);
	}
	if (_kbhit())
	{
		_getch();
	}
	SetPortVal(BASE_ADDRESS + iPort, 0x00, 1);
	ShutdownWinIo();
	return 0;
}
